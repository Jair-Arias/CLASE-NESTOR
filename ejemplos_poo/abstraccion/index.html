🔮 Abstracción en JavaScript

La abstracción oculta la complejidad interna y muestra solo las funcionalidades esenciales.

Ejemplo: Sistema de Dispositivos Electrónicos

// Clase abstracta (simulada) class DispositivoElectronico { constructor(marca, modelo) { if (this.constructor === DispositivoElectronico) { throw new Error("No se puede instanciar una clase abstracta"); } this.marca = marca; this.modelo = modelo; this.encendido = false; } // Métodos abstractos (deben ser implementados por las clases hijas) encender() { throw new Error("Método 'encender' debe ser implementado"); } apagar() { throw new Error("Método 'apagar' debe ser implementado"); } // Método concreto (implementación común) obtenerInfo() { return `${this.marca} ${this.modelo}`; } } // Implementaciones concretas class Television extends DispositivoElectronico { constructor(marca, modelo, pulgadas) { super(marca, modelo); this.pulgadas = pulgadas; this.canal = 1; this.volumen = 20; } encender() { this.encendido = true; return `📺 TV ${this.obtenerInfo()} encendida. Pantalla: ${this.pulgadas}"`; } apagar() { this.encendido = false; return `📺 TV ${this.obtenerInfo()} apagada`; } cambiarCanal(canal) { if (!this.encendido) return "La TV está apagada"; this.canal = canal; return `Canal cambiado a ${canal}`; } } class Smartphone extends DispositivoElectronico { constructor(marca, modelo, sistemaOperativo) { super(marca, modelo); this.sistemaOperativo = sistemaOperativo; this.bateria = 100; } encender() { this.encendido = true; return `📱 ${this.obtenerInfo()} encendido. SO: ${this.sistemaOperativo}`; } apagar() { this.encendido = false; return `📱 ${this.obtenerInfo()} apagado`; } llamar(numero) { if (!this.encendido) return "El teléfono está apagado"; return `Llamando al número ${numero}...`; } }
