⚡ Sobrecarga de Métodos en JavaScript

La sobrecarga permite que un método se comporte de manera diferente según los parámetros que recibe.

Ejemplo: Calculadora con Métodos Sobrecargados

class Calculadora { // Método sumar sobrecargado sumar(a, b, c) { if (c !== undefined) { return a + b + c; } else if (b !== undefined) { return a + b; } else if (Array.isArray(a)) { return a.reduce((total, num) => total + num, 0); } else { return a; } } // Método multiplicar sobrecargado multiplicar(a, b, c) { if (c !== undefined) { return a * b * c; } else if (b !== undefined) { return a * b; } else if (Array.isArray(a)) { return a.reduce((total, num) => total * num, 1); } else { return a; } } // Método potencia sobrecargado potencia(base, exponente) { if (exponente === undefined) { return base * base; // Cuadrado por defecto } return Math.pow(base, exponente); } } // Uso de los métodos sobrecargados: const calc = new Calculadora(); console.log(calc.sumar(5)); // 5 console.log(calc.sumar(5, 3)); // 8 console.log(calc.sumar(5, 3, 2)); // 10 console.log(calc.sumar([1,2,3,4])); // 10
